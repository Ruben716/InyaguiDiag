# Recurso de version de Windows.
#
# Sin esto el .exe aparece sin datos en las propiedades del archivo, lo
# que en una herramienta que se lleva a maquinas ajenas se ve mal y
# ademas aumenta los falsos positivos de los antivirus: un ejecutable sin
# metadatos ni firma es exactamente el perfil que buscan.

VSVersionInfo(
  ffi=FixedFileInfo(
    filevers=(0, 1, 0, 0),
    prodvers=(0, 1, 0, 0),
    mask=0x3f,
    flags=0x0,
    OS=0x40004,
    fileType=0x1,
    subtype=0x0,
    date=(0, 0)
  ),
  kids=[
    StringFileInfo([
      StringTable(
        '0c0a04b0',
        [StringStruct('CompanyName', 'Inyagui Solutions'),
         StringStruct('FileDescription', 'InyaguiDiag - Diagnostico de sistema'),
         StringStruct('FileVersion', '0.1.0'),
         StringStruct('InternalName', 'InyaguiDiag'),
         StringStruct('LegalCopyright', 'Inyagui Solutions'),
         StringStruct('OriginalFilename', 'InyaguiDiag.exe'),
         StringStruct('ProductName', 'InyaguiDiag'),
         StringStruct('ProductVersion', '0.1.0')])
    ]),
    VarFileInfo([VarStruct('Translation', [0x0c0a, 1200])])
  ]
)
